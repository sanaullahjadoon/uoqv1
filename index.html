یونٹ 4 : سبق 1 - (هُوَ)

قاعدہ: هُوَ کا معنی 'وہ' ہے اور یہ کسی شخص یا چیز کے اصل نام کی جگہ استعمال ہوتا ہے۔

مثال ۲رَسُوْلٌرسول ہےهُوَ رَسُوْلٌوہ رسول ہے۔
مثال ۳شِفَاءٌشفاء ہےهُوَ شِفَاءٌوہ شفاء ہے۔

قرآنی قطعات و آیات (صفحہ 3 - 5)

نمبرقرآنی جملہترجمہ
1هُوَ إِلَهٌ وَاحِدٌوہ ایک معبود ہے
2هُوَ شِفَاءٌ وَرَحْمَةٌوہ شفاء اور رحمت ہے
3هُوَ قُرْآنٌ مَجِيدٌوہ بڑا مجید قرآن ہے
4هُوَ نَبَأٌ عَظِيمٌوہ ایک بڑی خبر ہے
5هُوَ فِي ضَلَالٍ مُبِينٍوہ کھلی گمراہی میں ہے
6هُوَ مُسْرِفٌ كَذَّابٌوہ حد سے بڑھنے والا جھوٹا ہے
7هُوَ خَالِدٌ فِي النَّارِوہ آگ میں ہمیشہ رہنے والا ہے
8فَهُوَ عَلَى نُورٍپس وہ نور پر ہے
9هُوَ مِنَ الْكِتَابِوہ کتاب میں سے ہے
10هُوَ أَقْرَبُ لِلتَّقْوَىوہ تقویٰ کے زیادہ قریب ہے
11إِنَّمَا هُوَ إِلَهٌ وَاحِدٌبس وہی ایک معبود ہے
12هُوَ عَالِمُ الْغَيْبِ وَالشَّهَادَةِوہ غیب اور ظاہر کا جاننے والا ہے
13وَهُوَ عِنْدَ اللَّهِ عَظِيمٌاور وہ اللہ کے نزدیک بڑا ہے
14فَهُوَ فِي الْآخِرَةِ أَعْمَىپس وہ آخرت میں اندھا ہوگا
15هُوَ اللَّهُ أَحَدٌوہ اللہ ایک ہے
16هُوَ أَهْلُ التَّقْوَى وَأَهْلُ الْمَغْفِرَةِوہی تقویٰ کا اہل اور مغفرت کا مالک ہے
17وَهُوَ رَبُّ كُلِّ شَيْءٍاور وہ ہر چیز کا رب ہے
18وَهُوَ عَلَى كُلِّ شَيْءٍ قَدِيرٌاور وہ ہر چیز پر قادر ہے
19وَهُوَ عَلَى كُلِّ شَيْءٍ شَهِيدٌاور وہ ہر چیز پر گواہ ہے
20وَهُوَ عَلَى كُلِّ شَيْءٍ وَكِيلٌاور وہ ہر چیز پر نگہبان/نگران ہے
21وَهُوَ بِكُلِّ شَيْءٍ عَلِيمٌاور وہ ہر چیز کا جاننے والا ہے
22وَهُوَ بِكُلِّ خَلْقٍ عَلِيمٌاور وہ ہر مخلوق کو خوب جاننے والا ہے
23فَهُوَ عَلَى كُلِّ شَيْءٍ قَدِيرٌپس وہ ہر چیز پر قادر ہے
24هُوَ مِنْ عِنْدِ اللَّهِوہ اللہ کے پاس سے ہے
25هُوَ اللَّهُ الْعَزِيزُ الْحَكِيمُوہی اللہ غالب، حکمت والا ہے
26هُوَ اللَّهُ الْوَاحِدُ الْقَهَّارُوہی اللہ ایک ہے سب پر غالب ہے
27وَهُوَ اللَّهُ فِي السَّمَاوَاتِ وَفِي الْأَرْضِاور وہی اللہ ہے آسمانوں اور زمین میں
28لَا إِلَهَ إِلَّا هُوَنہیں کوئی معبود مگر وہی
29هُوَ قُرْآنٌ مَجِيدٌ فِي لَوْحٍ مَحْفُوظٍوہ قرآن مجید ہے لوح محفوظ میں
30فَهُوَ فِي عِيشَةٍ رَاضِيَةٍ فِي جَنَّةٍ عَالِيَةٍپس وہ من پسند زندگی میں ہوگا اونچی جنت میں
31اللَّهُ خَالِقُ كُلِّ شَيْءٍ وَهُوَ عَلَى كُلِّ شَيْءٍ وَكِيلٌاللہ ہر چیز کا خالق ہے اور وہ ہر چیز پر وکیل/نگران ہے
32وَلِلَّهِ غَيْبُ السَّمَاوَاتِ وَالْأَرْضِ وَمَا أَمْرُ السَّاعَةِ إِلَّا كَلَمْحِ الْبَصَرِ أَوْ هُوَ أَقْرَبُ إِنَّ اللَّهَ عَلَى كُلِّ شَيْءٍ قَدِيرٌاور اللہ کے لیے ہے غیب آسمانوں اور زمین کا اور نہیں معاملہ قیامت کا مگر آنکھ جھپکنے کی طرح یا وہ اس سے بھی زیادہ قریب ہے یقیناً اللہ ہر چیز پر قادر ہے

تاکیدی تعارف (الـ کے ساتھ): اگر هُوَ کے بعد الـ والا لفظ آئے تو تاکید (وہی) کا مفہوم پیدا ہوتا ہے۔

صفات الٰہیہ اور تاکیدی ترکیبیں (صفحہ 4 - 5)

1هُوَ التَّوَّابُ الرَّحِيمُوہی توبہ قبول کرنے والا، رحم کرنے والا ہے
2هُوَ الرَّحْمَنُ الرَّحِيمُوہی رحمن اور رحیم ہے
3هُوَ السَّمِيعُ الْبَصِيرُوہی سننے والا دیکھنے والا ہے
4وَهُوَ السَّمِيعُ الْعَلِيمُاور وہی سننے والا جاننے والا ہے
5هُوَ الْحَيُّ الْقَيُّومُوہی زندہ اور قائم رکھنے والا ہے
6هُوَ الْعَزِيزُ الْغَفَّارُوہی غالب اور بخشنے والا ہے
7وَهُوَ الْفَتَّاحُ الْعَلِيمُاور وہی سب سے بڑا فیصلہ کرنے والا، جاننے والا ہے
8وَهُوَ الْحَكِيمُ الْعَلِيمُاور وہی حکمت والا، علم والا ہے

11وَهُوَ الْعَزِيزُ الْغَفُورُاور وہی غالب، بخشنے والا ہے
12هُوَ الْعَزِيزُ الْحَكِيمُوہی غالب، حکمت والا ہے
13وَهُوَ الْعَلِيُّ الْكَبِيرُاور وہی عالی شان، بڑا ہے
14وَهُوَ الْعَلِيُّ الْعَظِيمُاور وہی بلند مرتبہ، عظمت والا ہے
15هُوَ الْأَوَّلُ وَالْآخِرُوہی اول اور وہی آخر ہے
16وَهُوَ الْخَلَّاقُ الْعَلِيمُاور وہی پیدا کرنے والا، جاننے والا ہے
17وَهُوَ الْوَاحِدُ الْقَهَّارُاور وہی ایک، سب پر غالب ہے
18هُوَ الرَّزَّاقُ ذُو الْقُوَّةِوہی رزاق، قوت والا ہے
19ذَلِكَ هُوَ الْفَوْزُ الْعَظِيمُیہی ہے بڑی کامیابی
20وَذَلِكَ هُوَ الْفَوْزُ الْعَظِيمُاور یہی ہے بڑی کامیابی
21ذَلِكَ هُوَ الْفَضْلُ الْكَبِيرُیہ بہت بڑا فضل ہے
22ذَلِكَ هُوَ الْفَوْزُ الْمُبِينُیہ کھلی کامیابی ہے
23ذَلِكَ هُوَ الضَّلَالُ الْبَعِيدُیہ دور کی گمراہی ہے
24ذَلِكَ هُوَ الْخُسْرَانُ الْمُبِينُیہ صریح خسارہ ہے
25وَاللَّهُ هُوَ السَّمِيعُ الْعَلِيمُاور اللہ ہی سننے والا، جاننے والا ہے

28وَأَنَّ اللَّهَ هُوَ الْعَلِيُّ الْكَبِيرُاور بیشک اللہ ہی بلند اور بڑا ہے
29وَإِنَّ اللَّهَ لَهُوَ الْعَزِيزُ الْحَكِيمُاور بیشک اللہ، وہی غالب، حکمت والا ہے

مزید جملے (صفحہ 5)

نمبرقرآنی جملہترجمہ
1فَإِنَّ أَجَلَ اللَّهِ لَآتٍ وَهُوَ السَّمِيعُ الْعَلِيمُپس بیشک اللہ کا مقررہ وقت یقیناً آنے والا ہے اور وہی سننے والا، جاننے والا ہے
2لِلَّهِ مَا فِي السَّمَاوَاتِ وَالْأَرْضِ إِنَّ اللَّهَ هُوَ الْغَنِيُّ الْحَمِيدُاللہ ہی کے لیے ہے جو کچھ آسمانوں اور زمین میں ہے، بیشک اللہ ہی غنی، تعریف کیا ہوا ہے
3هُوَ الْأَوَّلُ وَالْآخِرُ وَالظَّاهِرُ وَالْبَاطِنُ وَهُوَ بِكُلِّ شَيْءٍ عَلِيمٌوہی اول ہے اور آخر ہے اور ظاہر ہے اور باطن ہے اور وہ ہر چیز کا جاننے والا ہے
4وَهُوَ الَّذِي فِي السَّمَاءِ إِلَهٌ وَفِي الْأَرْضِ إِلَهٌ وَهُوَ الْحَكِيمُ الْعَلِيمُاور وہی ہے جو آسمان میں بھی معبود ہے اور زمین میں بھی معبود ہے اور وہی حکمت والا، جاننے والا ہے

1فَإِنَّمَا حِسَابُهُ عِنْدَ رَبِّهِپس اس کا حساب تو بس اس کے رب کے پاس ہے
2فَهُوَ عَلَى نُورٍ مِنْ رَبِّهِپس وہ اپنے رب کی طرف سے نور پر ہے
3فِي قَلْبِهِ مَرَضٌاس کے دل میں بیماری ہے
4إِنَّ أَخْذَهُ أَلِيمٌ شَدِيدٌبیشک اس کی پکڑ دردناک اور سخت ہے
5وَهُوَ ظَالِمٌ لِنَفْسِهِاور وہ اپنی جان پر ظلم کرنے والا ہے
6وَلِلَّهِ الْعِزَّةُ وَلِرَسُولِهِاور اللہ ہی کے لیے عزت ہے اور اس کے رسول کے لیے
7كَانَ عَلَى بَيِّنَةٍ مِنْ رَبِّهِوہ اپنے رب کی طرف سے روشن دلیل پر تھا
8وَاللَّهُ مُتِمُّ نُورِهِاور اللہ اپنے نور کو پورا کرنے والا ہے
9وَهَذَا مِنْ عَدُوِّهِاور یہ اس کے دشمن میں سے تھا
10هَذَا مِنْ شِيعَتِهِیہ اس کے گروہ میں سے ہے
11فَلِأُمِّهِ الثُّلُثُتو اس کی ماں کے لیے تیسرا حصہ ہے
12ضُرُّهُ أَقْرَبُ مِنْ نَفْعِهِاس کا نقصان اس کے نفع سے زیادہ قریب ہے
13أَنَّ صَالِحًا مُرْسَلٌ مِنْ رَبِّهِبیشک صالح اپنے رب کی طرف سے بھیجے گئے ہیں
14وَالنُّجُومُ مُسَخَّرَاتٌ بِأَمْرِهِاور ستارے اس کے حکم سے مسخر ہیں
15لَا مُبَدِّلَ لِكَلِمَاتِهِاس کے کلمات کو کوئی بدلنے والا نہیں
16وَإِنَّ مِنْ شِيعَتِهِ لَإِبْرَاهِيمَاور بیشک ابراہیم بھی اس کے گروہ میں سے تھے
17بِيَدِهِ الْمُلْكُاس کے ہاتھ میں بادشاہت ہے
18بِيَدِهِ مَلَكُوتُ كُلِّ شَيْءٍاس کے ہاتھ میں ہر چیز کی بادشاہت ہے
19إِنَّ الْإِنْسَانَ لِرَبِّهِ لَكَنُودٌبیشک انسان اپنے رب کا بڑا ناشکرا ہے
20وَكَانَ الشَّيْطَانُ لِرَبِّهِ كَفُورًااور شیطان اپنے رب کا بڑا ناشکرا تھا
21وَمِنْ وَرَائِهِ عَذَابٌ غَلِيظٌاور اس کے پیچھے سخت عذاب ہے
22وَكَانَ الْكَافِرُ عَلَى رَبِّهِ ظَهِيرًااور کافر اپنے رب کے خلاف مددگار ہے
23بَرَاءَةٌ مِنَ اللَّهِ وَرَسُولِهِبیزاری کا اعلان ہے اللہ اور اس کے رسول کی طرف سے
24وَأَذَانٌ مِنَ اللَّهِ وَرَسُولِهِاور اعلان ہے اللہ اور اس کے رسول کی طرف سے
25بِيَدِهِ الْمُلْكُ وَهُوَ عَلَى كُلِّ شَيْءٍ قَدِيرٌاس کے ہاتھ میں بادشاہت ہے اور وہ ہر چیز پر قادر ہے
26فَسُبْحَانَ الَّذِي بِيَدِهِ مَلَكُوتُ كُلِّ شَيْءٍپس پاک ہے وہ ذات جس کے ہاتھ میں ہر چیز کی بادشاہت ہے
27لَا مُبَدِّلَ لِكَلِمَاتِهِ وَهُوَ السَّمِيعُ الْعَلِيمُاس کے کلمات کو کوئی بدلنے والا نہیں اور وہی سننے والا جاننے والا ہے
28وَمِنْ آيَاتِهِ خَلْقُ السَّمَاوَاتِ وَالْأَرْضِاور اس کی نشانیوں میں سے ہے آسمانوں اور زمین کی پیدائش
29وَمِنْ آيَاتِهِ اللَّيْلُ وَالنَّهَارُ وَالشَّمْسُ وَالْقَمَرُاور اس کی نشانیوں میں سے ہیں رات اور دن اور سورج اور چاند
30بَلِ الْإِنْسَانُ عَلَى نَفْسِهِ بَصِيرَةٌبلکہ انسان خود اپنے نفس پر گواہ ہے
31هَذَا مِنْ شِيعَتِهِ وَهَذَا مِنْ عَدُوِّهِیہ اس کے گروہ میں سے ہے اور یہ اس کے دشمن میں سے ہے
32لَا مُعَقِّبَ لِحُكْمِهِ وَهُوَ سَرِيعُ الْحِسَابِاس کے حکم کو کوئی موخر کرنے/روکنے والا نہیں اور وہ جلد حساب لینے والا ہے
33وَالنُّجُومُ مُسَخَّرَاتٌ بِأَمْرِهِ إِنَّ فِي ذَلِكَ لَآيَاتٍاور ستارے اس کے حکم سے مسخر ہیں، بیشک اس میں نشانیاں ہیں
34وَأَذَانٌ مِنَ اللَّهِ وَرَسُولِهِ إِلَى النَّاسِ يَوْمَ الْحَجِّ الْأَكْبَرِ أَنَّ اللَّهَ بَرِيءٌاور اعلان ہے اللہ اور اس کے رسول کی طرف سے لوگوں کی طرف حجِ اکبر کے دن کہ بیشک اللہ بیزار ہے

س 4: درج ذیل قرآنی جملوں کا مطلب بتائیے:
1. هِيَ ظَالِمَةٌ ← وہ ظالم ہے۔

3. وَإِنَّ الْآخِرَةَ هِيَ دَارُ الْقَرَارِ ← اور بیشک آخرت ہی قائم رہنے کا گھر ہے۔
4. إِنَّمَا عِلْمُهَا عِنْدَ اللهِ ← اس کا علم تو صرف اللہ ہی کے پاس ہے۔
5. وَفِي نُسْخَتِهَا هُدًى وَرَحْمَةٌ ← اور اس کے نسخے میں ہدایت اور رحمت ہے۔
6. وَلَهَا عَرْشٌ عَظِيمٌ ← اور اس کے لیے ایک بڑا (عظیم) تخت ہے۔
7. فَهُوَ فِي الْآخِرَةِ أَعْمَى ← پس وہ آخرت میں (بھی) اندھا ہو گا۔
8. فَهُوَ فِي عِيشَةٍ رَّاضِيَةٍ ← پس وہ پسندیدہ/خوشگوار زندگی میں ہو گا۔
9. فَإِنَّ أَجَلَ اللهِ لَآتٍ ← پس بیشک اللہ کا مقرر کردہ وقت یقیناً آنے والا ہے۔
10. فَجَزَاؤُهُ جَهَنَّمُ ← پس اس کا بدلہ جہنم ہے۔
11. إِنَّ رَبَّكَ لَذُو مَغْفِرَةٍ ← بیشک تیرا رب یقیناً مغفرت والا ہے۔
12. إِنَّ هَذَا عَدُوٌّ لَّكَ وَلِزَوْجِكَ ← بیشک یہ تیرا اور تیری بیوی کا دشمن ہے۔
13. وَرِزْقُ رَبِّكَ خَيْرٌ وَأَبْقَى ← اور تیرے رب کا رزق بہتر اور باقی رہنے والا ہے۔
14. وَإِنَّ عَلَيْكَ لَعْنَتِي إِلَى يَوْمِ الدِّينِ ← اور بیشک تجھ پر جزا و سزا (قیامت) کے دن تک میری لعنت ہے۔
15. وَقَلْبُهُ مُطْمَئِنٌّ بِالْإِيمَانِ ← جبکہ اس کا دل ایمان پر مطمئن ہو۔
16. فَهُوَ عَلَى نُورٍ مِّن رَّبِّهِ ← پس وہ اپنے رب کی طرف سے ایک نور پر ہے۔
17. لَا مُبَدِّلَ لِكَلِمَاتِهِ ← اس کے کلمات کو کوئی بدلنے والا نہیں۔
18. وَمِنْ آيَاتِهِ اللَّيْلُ وَالنَّهَارُ ← اور رات اور دن اس کی نشانیوں میں سے ہیں۔
19. وَكَانَ رَبُّكَ قَدِيرًا ← اور تیرا رب ہر چیز پر قدرت رکھنے والا ہے۔
